Master Native Auto-Weight Skinning Engine Script cat << 'EOF' > master_rig_engine.py import bpy import mathutils import math import sys print("--- STARTING NATIVE AUTO-WEIGHT SKINNING ENGINE ---") bpy.ops.wm.read_factory_settings(use_empty=True) # 1. Import source model try: bpy.ops.import_scene.gltf(filepath='source.glb') print("✅ Imported source GLB model") except Exception as e: print(f"❌ Failed to import glTF: {e}") sys.exit(1) # Remove scene lights and cameras for obj in list(bpy.context.scene.objects): if obj.type in ['CAMERA', 'LIGHT']: bpy.data.objects.remove(obj, do_unlink=True) # 2. BAKE EVALUATED WORLD MATRICES ONTO VERTICES bpy.context.view_layer.update() depsgraph = bpy.context.evaluated_depsgraph_get() mesh_objs = [o for o in bpy.context.scene.objects if o.type == 'MESH'] if not mesh_objs: print("❌ Error: No mesh found.") sys.exit(1) new_mesh_objs = [] for obj in mesh_objs: eval_obj = obj.evaluated_get(depsgraph) me = bpy.data.meshes.new_from_object(eval_obj) me.transform(eval_obj.matrix_world) new_obj = bpy.data.objects.new(f"Clean_{obj.name}", me) for mat in obj.data.materials: new_obj.data.materials.append(mat) bpy.context.scene.collection.objects.link(new_obj) new_mesh_objs.append(new_obj) for obj in list(bpy.context.scene.objects): if obj not in new_mesh_objs: bpy.data.objects.remove(obj, do_unlink=True) # Join all parts (head, body, hair, clothing) into 1 unified mesh bpy.ops.object.select_all(action='DESELECT') for o in new_mesh_objs: o.select_set(True) bpy.context.view_layer.objects.active = new_mesh_objs[0] bpy.ops.object.join() character = bpy.context.active_object character.name = "HERO_AR_Mesh" # Rotate upright if lying horizontal bbox = [v.co for v in character.data.vertices] dim_y = max(v.y for v in bbox) - min(v.y for v in bbox) dim_z = max(v.z for v in bbox) - min(v.z for v in bbox) if dim_y > dim_z: print("🔄 Rotating character upright (+90° X)...") rot_mat = mathutils.Matrix.Rotation(math.radians(90), 4, 'X') character.data.transform(rot_mat) character.data.update() # Center pivot point to mesh bounds and snap location to World Origin (0,0,0) bpy.ops.object.origin_set(type='ORIGIN_GEOMETRY', center='BOUNDS') character.location = (0.0, 0.0, 0.0) character.rotation_euler = (0.0, 0.0, 0.0) character.scale = (1.0, 1.0, 1.0) bpy.ops.object.transform_apply(location=True, rotation=True, scale=True) # Weld neck seam vertices without destroying geometry bpy.ops.object.mode_set(mode='EDIT') bpy.ops.mesh.select_all(action='SELECT') bpy.ops.mesh.remove_doubles(threshold=0.001) bpy.ops.mesh.normals_make_consistent(inside=False) bpy.ops.object.mode_set(mode='OBJECT') for poly in character.data.polygons: poly.use_smooth = True # Pack PBR textures for img in bpy.data.images: if img.has_data and img.name != 'Render Result': try: img.pack() except Exception: pass # 3. MEASURE MESH GEOMETRY BOUNDS verts = [v.co for v in character.data.vertices] min_x, max_x = min(v.x for v in verts), max(v.x for v in verts) min_y, max_y = min(v.y for v in verts), max(v.y for v in verts) min_z, max_z = min(v.z for v in verts), max(v.z for v in verts) height = max_z - min_z width = max_x - min_x depth = max_y - min_y cx = (min_x + max_x) / 2.0 cy = (min_y + max_y) / 2.0 # Detect outermost hand coordinates on mesh arm_zone = [v for v in verts if min_z + height * 0.35 <= v.z <= min_z + height * 0.85] if arm_zone: hand_l = max(arm_zone, key=lambda v: v.x) hand_r = min(arm_zone, key=lambda v: v.x) else: hand_l = mathutils.Vector((cx + width * 0.45, cy, min_z + height * 0.5)) hand_r = mathutils.Vector((cx - width * 0.45, cy, min_z + height * 0.5)) shoulder_z = min_z + height * 0.78 clav_l = mathutils.Vector((cx + (hand_l.x - cx) * 0.22, hand_l.y, shoulder_z)) clav_r = mathutils.Vector((cx + (hand_r.x - cx) * 0.22, hand_r.y, shoulder_z)) elbow_l = clav_l.lerp(hand_l, 0.50) wrist_l = clav_l.lerp(hand_l, 0.85) elbow_r = clav_r.lerp(hand_r, 0.50) wrist_r = clav_r.lerp(hand_r, 0.85) # 4. CONSTRUCT COMPLETE SKELETON (SPINE, FACE, HAIR, ARMS, FINGERS, LEGS) arm_data = bpy.data.armatures.new("ArmatureData") rig = bpy.data.objects.new("Armature", arm_data) bpy.context.scene.collection.objects.link(rig) rig.location = (0.0, 0.0, 0.0) rig.rotation_euler = (0.0, 0.0, 0.0) rig.scale = (1.0, 1.0, 1.0) bpy.context.view_layer.objects.active = rig bpy.ops.object.mode_set(mode='EDIT') bones_def = [ # Spine Column ("Hips", (cx, cy, min_z + height * 0.48), (cx, cy, min_z + height * 0.60), None), ("Spine", (cx, cy, min_z + height * 0.60), (cx, cy, min_z + height * 0.73), "Hips"), ("Chest", (cx, cy, min_z + height * 0.73), (cx, cy, min_z + height * 0.82), "Spine"), ("Neck", (cx, cy, min_z + height * 0.82), (cx, cy, min_z + height * 0.87), "Chest"), ("Head", (cx, cy, min_z + height * 0.87), (cx, cy, max_z), "Neck"), # Face & Hair ("Jaw", (cx, cy - depth * 0.05, min_z + height * 0.87), (cx, cy - depth * 0.15, min_z + height * 0.83), "Head"), ("Eye_L", (cx + width * 0.05, cy - depth * 0.14, min_z + height * 0.90), (cx + width * 0.05, cy - depth * 0.18, min_z + height * 0.90), "Head"), ("Eye_R", (cx - width * 0.05, cy - depth * 0.14, min_z + height * 0.90), (cx - width * 0.05, cy - depth * 0.18, min_z + height * 0.90), "Head"), ("Hair_Root", (cx, cy, min_z + height * 0.94), (cx, cy, max_z + height * 0.02), "Head"), # Arms ("Clavicle_L", (cx, cy, shoulder_z), clav_l, "Chest"), ("UpperArm_L", clav_l, elbow_l, "Clavicle_L"), ("Forearm_L", elbow_l, wrist_l, "UpperArm_L"), ("Hand_L", wrist_l, hand_l, "Forearm_L"), ("Clavicle_R", (cx, cy, shoulder_z), clav_r, "Chest"), ("UpperArm_R", clav_r, elbow_r, "Clavicle_R"), ("Forearm_R", elbow_r, wrist_r, "UpperArm_R"), ("Hand_R", wrist_r, hand_r, "Forearm_R"), # Legs ("Thigh_L", (cx + width * 0.12, cy, min_z + height * 0.48), (cx + width * 0.12, cy, min_z + height * 0.25), "Hips"), ("Shin_L", (cx + width * 0.12, cy, min_z + height * 0.25), (cx + width * 0.12, cy, min_z + height * 0.05), "Thigh_L"), ("Foot_L", (cx + width * 0.12, cy, min_z + height * 0.05), (cx + width * 0.12, cy - depth * 0.18, min_z), "Shin_L"), ("Thigh_R", (cx - width * 0.12, cy, min_z + height * 0.48), (cx - width * 0.12, cy, min_z + height * 0.25), "Hips"), ("Shin_R", (cx - width * 0.12, cy, min_z + height * 0.25), (cx - width * 0.12, cy, min_z + height * 0.05), "Thigh_R"), ("Foot_R", (cx - width * 0.12, cy, min_z + height * 0.05), (cx - width * 0.12, cy - depth * 0.18, min_z), "Shin_R") ] # 30 Finger Joint Bones finger_configs = [("Thumb", 0.02, -0.01), ("Index", 0.01, -0.02), ("Middle", 0.00, -0.025), ("Ring", -0.01, -0.02), ("Pinky", -0.02, -0.015)] for side, sign, h_pos, parent_hand in [("L", 1.0, hand_l, "Hand_L"), ("R", -1.0, hand_r, "Hand_R")]: hx, hy, hz = h_pos.x, h_pos.y, h_pos.z for f_name, offset_y_scale, dz_scale in finger_configs: offset_y = offset_y_scale * depth b1, b2, b3 = f"{f_name}1_{side}", f"{f_name}2_{side}", f"{f_name}3_{side}" p1 = (hx, hy + offset_y, hz) p2 = (hx + sign * width * 0.015, hy + offset_y, hz + dz_scale * height * 0.5) p3 = (hx + sign * width * 0.028, hy + offset_y, hz + dz_scale * height * 1.0) p4 = (hx + sign * width * 0.040, hy + offset_y, hz + dz_scale * height * 1.5) bones_def.append((b1, p1, p2, parent_hand)) bones_def.append((b2, p2, p3, b1)) bones_def.append((b3, p3, p4, b2)) created = {} for name, hp, tp, p_name in bones_def: b = arm_data.edit_bones.new(name) b.head, b.tail = hp, tp created[name] = b for name, hp, tp, p_name in bones_def: if p_name and p_name in created: created[name].parent = created[p_name] bpy.ops.object.mode_set(mode='OBJECT') # 5. BIND VERTICES USING BLENDER NATIVE AUTOMATIC HEAT WEIGHTS (ARMATURE_AUTO) bpy.ops.object.select_all(action='DESELECT') character.select_set(True) rig.select_set(True) bpy.context.view_layer.objects.active = rig try: print("⚡ Calculating native automatic skinning heat weights...") bpy.ops.object.parent_set(type='ARMATURE_AUTO') print("✅ ARMATURE_AUTO bound successfully") except Exception as e: print(f"⚠️ Falling back to distance skinning solver: {e}") character.vertex_groups.clear() vg_map = {b_name: character.vertex_groups.new(name=b_name) for b_name, _, _, _ in bones_def} for v in character.data.vertices: v_co = v.co closest_bone = None min_dist = float('inf') for b_name, h_pos, t_pos, _ in bones_def: p1 = mathutils.Vector(h_pos) p2 = mathutils.Vector(t_pos) line_vec = p2 - p1 line_len = line_vec.length dist = (v_co - p1).length if line_len == 0 else (v_co - (p1 + max(0.0, min(1.0, (v_co - p1).dot(line_vec) / (line_len ** 2))) * line_vec)).length if dist < min_dist: min_dist = dist closest_bone = b_name if closest_bone: vg_map[closest_bone].add([v.index], 1.0, 'REPLACE') character.parent = rig arm_mod = character.modifiers.new(name="Armature", type='ARMATURE') arm_mod.object = rig # 6. EXPORT FINAL RIGGED GLB MASTER FILE print("💾 Exporting HERO_AR_RIGGED_MASTER.glb...") bpy.ops.export_scene.gltf( filepath='HERO_AR_RIGGED_MASTER.glb', export_format='GLB', export_materials='EXPORT', export_yup=True ) print("🎉 NATIVE AUTO-WEIGHT SKINNING COMPLETE!")
